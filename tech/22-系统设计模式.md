```typescript
// API Gateway 实现
class APIGateway {
    private routes: Map<string, ServiceConfig> = new Map();
    private middleware: Middleware[] = [];
    private rateLimiter: RateLimiter;
    private loadBalancer: LoadBalancer;
    
    constructor(config: GatewayConfig) {
        this.rateLimiter = new RateLimiter(config.rateLimit);
        this.loadBalancer = new LoadBalancer(config.services);
    }
    
    // 注册路由
    registerRoute(path: string, service: ServiceConfig) {
        this.routes.set(path, service);
    }
    
    // 使用中间件
    use(middleware: Middleware) {
        this.middleware.push(middleware);
    }
    
    // 处理请求
    async handleRequest(req: Request): Promise<Response> {
        // 执行中间件
        for (const mw of this.middleware) {
            const result = await mw(req);
            if (result) return result;
        }
        
        // 限流检查
        if (!this.rateLimiter.allow(req)) {
            return new Response('Too Many Requests', { status: 429 });
        }
        
        // 路由匹配
        const service = this.matchRoute(req.path);
        if (!service) {
            return new Response('Not Found', { status: 404 });
        }
        
        // 负载均衡
        const instance = this.loadBalancer.getInstance(service);
        
        // 转发请求
        return this.forwardRequest(req, instance);
    }
    
    private matchRoute(path: string): ServiceConfig | undefined {
        for (const [route, service] of this.routes) {
            if (path.startsWith(route)) {
                return service;
            }
        }
        return undefined;
    }
    
    private async forwardRequest(req: Request, instance: ServiceInstance): Promise<Response> {
        const url = `${instance.url}${req.path}`;
        const headers = new Headers(req.headers);
        headers.set('X-Forwarded-For', req.ip);
        headers.set('X-Request-ID', this.generateRequestId());
        
        return fetch(url, {
            method: req.method,
            headers,
            body: req.body
        });
    }
    
    private generateRequestId(): string {
        return `${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
    }
}

// 中间件类型
type Middleware = (req: Request) => Promise<Response | null>;

// 认证中间件
const authMiddleware: Middleware = async (req) => {
    const token = req.headers.get('Authorization');
    
    if (!token) {
        return new Response('Unauthorized', { status: 401 });
    }
    
    try {
        const user = await verifyToken(token);
        req.user = user;
        return null;
    } catch {
        return new Response('Invalid Token', { status: 401 });
    }
};

// 日志中间件
const loggingMiddleware: Middleware = async (req) => {
    const start = Date.now();
    console.log(`[${new Date().toISOString()}] ${req.method} ${req.path}`);
    
    // 继续处理
    return null;
};

// CORS 中间件
const corsMiddleware: Middleware = async (req) => {
    if (req.method === 'OPTIONS') {
        return new Response(null, {
            headers: {
                'Access-Control-Allow-Origin': '*',
                'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE',
                'Access-Control-Allow-Headers': 'Content-Type, Authorization'
            }
        });
    }
    return null;
};
```

```typescript
// 服务注册中心
class ServiceRegistry {
    private services: Map<string, ServiceInstance[]> = new Map();
    private healthChecks: Map<string, NodeJS.Timer> = new Map();
    
    // 注册服务
    register(serviceName: string, instance: ServiceInstance) {
        if (!this.services.has(serviceName)) {
            this.services.set(serviceName, []);
        }
        
        this.services.get(serviceName)!.push(instance);
        
        // 启动健康检查
        this.startHealthCheck(serviceName, instance);
        
        console.log(`Registered ${serviceName}: ${instance.url}`);
    }
    
    // 注销服务
    deregister(serviceName: string, instanceId: string) {
        const instances = this.services.get(serviceName);
        if (instances) {
            const index = instances.findIndex(i => i.id === instanceId);
            if (index > -1) {
                instances.splice(index, 1);
                console.log(`Deregistered ${serviceName}: ${instanceId}`);
            }
        }
    }
    
    // 获取服务实例
    getInstances(serviceName: string): ServiceInstance[] {
        return this.services.get(serviceName) || [];
    }
    
    // 获取健康实例
    getHealthyInstances(serviceName: string): ServiceInstance[] {
        const instances = this.getInstances(serviceName);
        return instances.filter(i => i.status === 'healthy');
    }
    
    // 健康检查
    private startHealthCheck(serviceName: string, instance: ServiceInstance) {
        const interval = setInterval(async () => {
            try {
                const response = await fetch(`${instance.url}/health`, {
                    timeout: 5000
                });
                
                if (response.ok) {
                    instance.status = 'healthy';
                    instance.lastCheck = Date.now();
                } else {
                    instance.status = 'unhealthy';
                    instance.failures = (instance.failures || 0) + 1;
                }
            } catch (error) {
                instance.status = 'unhealthy';
                instance.failures = (instance.failures || 0) + 1;
                
                // 连续失败3次，移除实例
                if (instance.failures >= 3) {
                    this.deregister(serviceName, instance.id);
                }
            }
        }, 30000); // 30秒检查一次
        
        this.healthChecks.set(instance.id, interval);
    }
}

// 服务实例
interface ServiceInstance {
    id: string;
    url: string;
    status: 'healthy' | 'unhealthy';
    lastCheck: number;
    failures: number;
    metadata: Record<string, any>;
}

// 客户端服务发现
class ServiceDiscovery {
    private registry: ServiceRegistry;
    private cache: Map<string, { instances: ServiceInstance[], expiry: number }> = new Map();
    
    constructor(registry: ServiceRegistry) {
        this.registry = registry;
    }
    
    async discover(serviceName: string): Promise<ServiceInstance> {
        // 检查缓存
        const cached = this.cache.get(serviceName);
        if (cached && cached.expiry > Date.now()) {
            return this.selectInstance(cached.instances);
        }
        
        // 从注册中心获取
        const instances = this.registry.getHealthyInstances(serviceName);
        
        if (instances.length === 0) {
            throw new Error(`No healthy instances for ${serviceName}`);
        }
        
        // 更新缓存
        this.cache.set(serviceName, {
            instances,
            expiry: Date.now() + 30000 // 30秒缓存
        });
        
        return this.selectInstance(instances);
    }
    
    private selectInstance(instances: ServiceInstance[]): ServiceInstance {
        // 轮询算法
        const index = Math.floor(Math.random() * instances.length);
        return instances[index];
    }
}
```

### 1.3 断路器模式
```typescript
// 断路器状态
enum CircuitState {
    CLOSED = 'CLOSED',      // 正常状态
    OPEN = 'OPEN',          // 熔断状态
    HALF_OPEN = 'HALF_OPEN' // 半开状态
}

// 断路器配置
interface CircuitBreakerConfig {
    failureThreshold: number;     // 失败阈值
    resetTimeout: number;         // 重置超时
    monitoringPeriod: number;     // 监控周期
    fallbackFunction?: Function;  // 降级函数
}

// 断路器实现
class CircuitBreaker {
    private state: CircuitState = CircuitState.CLOSED;
    private failureCount: number = 0;
    private lastFailureTime: number = 0;
    private config: CircuitBreakerConfig;
    
    constructor(config: CircuitBreakerConfig) {
        this.config = config;
    }
    
    async execute<T>(fn: () => Promise<T>): Promise<T> {
        // 检查断路器状态
        if (this.state === CircuitState.OPEN) {
            if (this.shouldReset()) {
                this.state = CircuitState.HALF_OPEN;
            } else {
                return this.fallback();
            }
        }
        
        try {
            const result = await fn();
            this.onSuccess();
            return result;
        } catch (error) {
            this.onFailure();
            throw error;
        }
    }
    
    private onSuccess() {
        this.failureCount = 0;
        this.state = CircuitState.CLOSED;
    }
    
    private onFailure() {
        this.failureCount++;
        this.lastFailureTime = Date.now();
        
        if (this.failureCount >= this.config.failureThreshold) {
            this.state = CircuitState.OPEN;
            console.log('Circuit breaker OPEN');
        }
    }
    
    private shouldReset(): boolean {
        return Date.now() - this.lastFailureTime >= this.config.resetTimeout;
    }
    
    private fallback(): any {
        if (this.config.fallbackFunction) {
            return this.config.fallbackFunction();
        }
        throw new Error('Circuit breaker is OPEN');
    }
    
    getState(): CircuitState {
        return this.state;
    }
}

// 使用示例
const breaker = new CircuitBreaker({
    failureThreshold: 5,
    resetTimeout: 60000,
    fallbackFunction: () => ({ message: 'Service temporarily unavailable' })
});

// 调用远程服务
const result = await breaker.execute(async () => {
    const response = await fetch('http://service/api/data');
    return response.json();
});
```

### 2.2 Event Sourcing 模式
```typescript
// 事件
interface DomainEvent {
    type: string;
    aggregateId: string;
    payload: any;
    timestamp: number;
    version: number;
}

// 事件存储
class EventStore {
    private events: DomainEvent[] = [];
    private snapshots: Map<string, any> = new Map();
    
    // 保存事件
    async saveEvent(event: DomainEvent): Promise<void> {
        this.events.push(event);
        console.log(`Event saved: ${event.type} for ${event.aggregateId}`);
    }
    
    // 获取事件流
    getEvents(aggregateId: string, fromVersion?: number): DomainEvent[] {
        return this.events.filter(e => 
            e.aggregateId === aggregateId &&
            (!fromVersion || e.version > fromVersion)
        );
    }
    
    // 保存快照
    async saveSnapshot(aggregateId: string, state: any, version: number): Promise<void> {
        this.snapshots.set(aggregateId, { state, version });
    }
    
    // 获取快照
    getSnapshot(aggregateId: string): { state: any; version: number } | undefined {
        return this.snapshots.get(aggregateId);
    }
}

// 聚合根
abstract class AggregateRoot {
    protected id: string;
    protected version: number = 0;
    private uncommittedEvents: DomainEvent[] = [];
    
    constructor(id: string) {
        this.id = id;
    }
    
    // 应用事件
    protected apply(event: DomainEvent): void {
        this.applyEvent(event);
        this.version++;
        this.uncommittedEvents.push(event);
    }
    
    // 应用事件到状态
    protected abstract applyEvent(event: DomainEvent): void;
    
    // 获取未提交事件
    getUncommittedEvents(): DomainEvent[] {
        return this.uncommittedEvents;
    }
    
    // 清除未提交事件
    clearUncommittedEvents(): void {
        this.uncommittedEvents = [];
    }
    
    // 从事件流重建
    static loadFromHistory<T extends AggregateRoot>(
        this: new (id: string) => T,
        id: string,
        events: DomainEvent[]
    ): T {
        const aggregate = new this(id);
        events.forEach(event => aggregate.applyEvent(event));
        aggregate.version = events.length;
        return aggregate;
    }
}

// 用户聚合
class UserAggregate extends AggregateRoot {
    private name: string = '';
    private email: string = '';
    private status: 'active' | 'inactive' = 'active';
    
    // 创建用户
    createUser(name: string, email: string): void {
        this.apply({
            type: 'UserCreated',
            aggregateId: this.id,
            payload: { name, email },
            timestamp: Date.now(),
            version: this.version + 1
        });
    }
    
    // 更新用户
    updateUser(name: string): void {
        this.apply({
            type: 'UserUpdated',
            aggregateId: this.id,
            payload: { name },
            timestamp: Date.now(),
            version: this.version + 1
        });
    }
    
    // 停用用户
    deactivateUser(): void {
        this.apply({
            type: 'UserDeactivated',
            aggregateId: this.id,
            payload: {},
            timestamp: Date.now(),
            version: this.version + 1
        });
    }
    
    // 应用事件
    protected applyEvent(event: DomainEvent): void {
        switch (event.type) {
            case 'UserCreated':
                this.name = event.payload.name;
                this.email = event.payload.email;
                break;
            case 'UserUpdated':
                this.name = event.payload.name;
                break;
            case 'UserDeactivated':
                this.status = 'inactive';
                break;
        }
    }
    
    // 获取状态
    getState() {
        return {
            id: this.id,
            name: this.name,
            email: this.email,
            status: this.status,
            version: this.version
        };
    }
}

// 事件存储库
class Repository<T extends AggregateRoot> {
    private eventStore: EventStore;
    private aggregateClass: new (id: string) => T;
    
    constructor(eventStore: EventStore, aggregateClass: new (id: string) => T) {
        this.eventStore = eventStore;
        this.aggregateClass = aggregateClass;
    }
    
    // 加载聚合
    async load(id: string): Promise<T> {
        // 检查快照
        const snapshot = this.eventStore.getSnapshot(id);
        let aggregate: T;
        let fromVersion = 0;
        
        if (snapshot) {
            aggregate = new this.aggregateClass(id);
            Object.assign(aggregate, snapshot.state);
            fromVersion = snapshot.version;
        }
        
        // 加载事件
        const events = this.eventStore.getEvents(id, fromVersion);
        aggregate = this.aggregateClass.loadFromHistory(id, events);
        
        return aggregate;
    }
    
    // 保存聚合
    async save(aggregate: T): Promise<void> {
        const events = aggregate.getUncommittedEvents();
        
        for (const event of events) {
            await this.eventStore.saveEvent(event);
        }
        
        aggregate.clearUncommittedEvents();
        
        // 定期保存快照
        if (aggregate.version % 10 === 0) {
            await this.eventStore.saveSnapshot(
                aggregate.id,
                aggregate.getState(),
                aggregate.version
            );
        }
    }
}

// 使用示例
const eventStore = new EventStore();
const userRepo = new Repository(eventStore, UserAggregate);

// 创建用户
const user = new UserAggregate('user-1');
user.createUser('John', 'john@example.com');
await userRepo.save(user);

// 加载用户
const loadedUser = await userRepo.load('user-1');
console.log(loadedUser.getState());
```

### 2.3 Saga 模式
```typescript
// Saga 步骤
interface SagaStep {
    name: string;
    execute: (context: SagaContext) => Promise<void>;
    compensate: (context: SagaContext) => Promise<void>;
}

// Saga 上下文
interface SagaContext {
    data: Map<string, any>;
    stepResults: Map<string, any>;
}

// Saga 协调器
class SagaOrchestrator {
    private steps: SagaStep[] = [];
    private completedSteps: SagaStep[] = [];
    
    // 添加步骤
    addStep(step: SagaStep): this {
        this.steps.push(step);
        return this;
    }
    
    // 执行 Saga
    async execute(context: SagaContext): Promise<void> {
        console.log('Starting saga execution');
        
        for (const step of this.steps) {
            try {
                console.log(`Executing step: ${step.name}`);
                await step.execute(context);
                this.completedSteps.push(step);
                console.log(`Step completed: ${step.name}`);
            } catch (error) {
                console.error(`Step failed: ${step.name}`, error);
                await this.compensate(context);
                throw error;
            }
        }
        
        console.log('Saga completed successfully');
    }
    
    // 补偿操作
    private async compensate(context: SagaContext): Promise<void> {
        console.log('Starting compensation');
        
        // 逆序执行补偿
        for (const step of this.completedSteps.reverse()) {
            try {
                console.log(`Compensating step: ${step.name}`);
                await step.compensate(context);
                console.log(`Compensation completed: ${step.name}`);
            } catch (error) {
                console.error(`Compensation failed: ${step.name}`, error);
                // 补偿失败需要人工介入
            }
        }
    }
}

// 使用示例：订单创建 Saga
const createOrderSaga = new SagaOrchestrator();

// 步骤1：创建订单
createOrderSaga.addStep({
    name: 'createOrder',
    execute: async (ctx) => {
        const order = await orderService.create(ctx.data.get('orderData'));
        ctx.stepResults.set('order', order);
    },
    compensate: async (ctx) => {
        const order = ctx.stepResults.get('order');
        await orderService.cancel(order.id);
    }
});

// 步骤2：预留库存
createOrderSaga.addStep({
    name: 'reserveInventory',
    execute: async (ctx) => {
        const order = ctx.stepResults.get('order');
        await inventoryService.reserve(order.items);
    },
    compensate: async (ctx) => {
        const order = ctx.stepResults.get('order');
        await inventoryService.release(order.items);
    }
});

// 步骤3：处理支付
createOrderSaga.addStep({
    name: 'processPayment',
    execute: async (ctx) => {
        const order = ctx.stepResults.get('order');
        await paymentService.charge(order.totalAmount);
    },
    compensate: async (ctx) => {
        const order = ctx.stepResults.get('order');
        await paymentService.refund(order.totalAmount);
    }
});

// 步骤4：发送确认邮件
createOrderSaga.addStep({
    name: 'sendConfirmation',
    execute: async (ctx) => {
        const order = ctx.stepResults.get('order');
        await emailService.send(order.customerEmail, 'Order Confirmed');
    },
    compensate: async (ctx) => {
        // 邮件无法撤回，记录日志
        console.log('Email cannot be revoked');
    }
});

// 执行 Saga
const context: SagaContext = {
    data: new Map([
        ['orderData', { items: [...], customerEmail: '...' }]
    ]),
    stepResults: new Map()
};

try {
    await createOrderSaga.execute(context);
    console.log('Order created successfully');
} catch (error) {
    console.error('Order creation failed, all changes rolled back');
}
```

### 3.1 Cache-Aside 模式
```typescript
class CacheAside {
    private cache: Map<string, { data: any; expiry: number }> = new Map();
    private ttl: number;
    
    constructor(ttl: number = 60000) {
        this.ttl = ttl;
    }
    
    async get<T>(key: string, fetchFn: () => Promise<T>): Promise<T> {
        // 检查缓存
        const cached = this.cache.get(key);
        if (cached && cached.expiry > Date.now()) {
            console.log(`Cache hit: ${key}`);
            return cached.data;
        }
        
        console.log(`Cache miss: ${key}`);
        
        // 从源获取数据
        const data = await fetchFn();
        
        // 存入缓存
        this.cache.set(key, {
            data,
            expiry: Date.now() + this.ttl
        });
        
        return data;
    }
    
    invalidate(key: string): void {
        this.cache.delete(key);
        console.log(`Cache invalidated: ${key}`);
    }
    
    invalidatePattern(pattern: RegExp): void {
        for (const key of this.cache.keys()) {
            if (pattern.test(key)) {
                this.cache.delete(key);
            }
        }
    }
}

// 使用示例
const cache = new CacheAside(30000); // 30秒 TTL

async function getUser(id: number): Promise<User> {
    return cache.get(`user:${id}`, async () => {
        return db.users.findById(id);
    });
}

async function updateUser(id: number, data: Partial<User>): Promise<User> {
    const user = await db.users.update(id, data);
    cache.invalidate(`user:${id}`);
    return user;
}
```

### 3.2 Write-Through 模式
```typescript
class WriteThrough {
    private cache: Map<string, any> = new Map();
    
    async read<T>(key: string, fetchFn: () => Promise<T>): Promise<T> {
        // 检查缓存
        if (this.cache.has(key)) {
            return this.cache.get(key);
        }
        
        // 从存储读取
        const data = await fetchFn();
        this.cache.set(key, data);
        return data;
    }
    
    async write<T>(key: string, data: T, saveFn: (data: T) => Promise<void>): Promise<void> {
        // 写入存储
        await saveFn(data);
        
        // 写入缓存
        this.cache.set(key, data);
    }
    
    async delete(key: string, deleteFn: () => Promise<void>): Promise<void> {
        // 从存储删除
        await deleteFn();
```
