## 一、ECS 架构（Entity-Component-System）

### 1.1 核心概念
```
传统 OOP：
GameObject
├── Player
│   ├── position
│   ├── health
│   └── render()
└── Enemy
    ├── position
    ├── health
    └── render()

问题：
- 继承层次深
- 代码复用困难
- 组合爆炸

ECS 解决方案：
Entity（实体）：只是一个 ID
Component（组件）：纯数据
System（系统）：纯逻辑

Entity 1: [Position, Velocity, Sprite]
Entity 2: [Position, Health, AI]
Entity 3: [Position, Velocity, Player]
```

```javascript
// Entity 只是一个 ID
class Entity {
    static nextId = 0;
    
    constructor() {
        this.id = Entity.nextId++;
        this.components = new Map();
    }
    
    addComponent(component) {
        this.components.set(component.constructor, component);
        return this;
    }
    
    removeComponent(ComponentType) {
        this.components.delete(ComponentType);
        return this;
    }
    
    getComponent(ComponentType) {
        return this.components.get(ComponentType);
    }
    
    hasComponent(ComponentType) {
        return this.components.has(ComponentType);
    }
}

// Component 是纯数据
class Position {
    constructor(x = 0, y = 0) {
        this.x = x;
        this.y = y;
    }
}

class Velocity {
    constructor(vx = 0, vy = 0) {
        this.vx = vx;
        this.vy = vy;
    }
}

class Health {
    constructor(current = 100, max = 100) {
        this.current = current;
        this.max = max;
    }
}

class Sprite {
    constructor(image, width, height) {
        this.image = image;
        this.width = width;
        this.height = height;
    }
}

// System 是纯逻辑
class MovementSystem {
    update(entities, dt) {
        entities.forEach(entity => {
            const pos = entity.getComponent(Position);
            const vel = entity.getComponent(Velocity);
            
            if (pos && vel) {
                pos.x += vel.vx * dt;
                pos.y += vel.vy * dt;
            }
        });
    }
}

class RenderSystem {
    constructor(ctx) {
        this.ctx = ctx;
    }
    
    update(entities) {
        entities.forEach(entity => {
            const pos = entity.getComponent(Position);
            const sprite = entity.getComponent(Sprite);
            
            if (pos && sprite) {
                this.ctx.drawImage(
                    sprite.image,
                    pos.x - sprite.width / 2,
                    pos.y - sprite.height / 2,
                    sprite.width,
                    sprite.height
                );
            }
        });
    }
}

// World 管理所有实体和系统
class World {
    constructor() {
        this.entities = new Map();
        this.systems = [];
    }
    
    createEntity() {
        const entity = new Entity();
        this.entities.set(entity.id, entity);
        return entity;
    }
    
    removeEntity(entity) {
        this.entities.delete(entity.id);
    }
    
    addSystem(system) {
        this.systems.push(system);
    }
    
    update(dt) {
        const entities = Array.from(this.entities.values());
        this.systems.forEach(system => system.update(entities, dt));
    }
    
    // 查询具有特定组件的实体
    query(...ComponentTypes) {
        return Array.from(this.entities.values()).filter(entity =>
            ComponentTypes.every(Type => entity.hasComponent(Type))
        );
    }
}
```

```javascript
// 位标记优化
class ComponentTypes {
    static nextBit = 0;
    static types = new Map();
    
    static register(ComponentClass) {
        const bit = this.nextBit++;
        this.types.set(ComponentClass, 1 << bit);
        return bit;
    }
    
    static getBit(ComponentClass) {
        return this.types.get(ComponentClass);
    }
}

// 注册组件
ComponentTypes.register(Position);
ComponentTypes.register(Velocity);
ComponentTypes.register(Health);

// Entity 使用位标记
class OptimizedEntity {
    constructor() {
        this.id = Entity.nextId++;
        this.componentBits = 0;
        this.components = new Map();
    }
    
    addComponent(component) {
        const bit = ComponentTypes.getBit(component.constructor);
        this.componentBits |= bit;
        this.components.set(component.constructor, component);
        return this;
    }
    
    hasComponent(ComponentType) {
        const bit = ComponentTypes.getBit(ComponentType);
        return (this.componentBits & bit) !== 0;
    }
    
    // 快速匹配
    matches(queryBits) {
        return (this.componentBits & queryBits) === queryBits;
    }
}

// System 使用预计算的查询位
class OptimizedMovementSystem {
    constructor() {
        this.queryBits = ComponentTypes.getBit(Position) | 
                         ComponentTypes.getBit(Velocity);
    }
    
    update(entities, dt) {
        entities.forEach(entity => {
            if (!entity.matches(this.queryBits)) return;
            
            const pos = entity.components.get(Position);
            const vel = entity.components.get(Velocity);
            
            pos.x += vel.vx * dt;
            pos.y += vel.vy * dt;
        });
    }
}
```

### 2.1 树形结构
```javascript
class SceneNode {
    constructor(name = '') {
        this.name = name;
        this.parent = null;
        this.children = [];
        
        // 变换
        this.localPosition = { x: 0, y: 0 };
        this.localRotation = 0;
        this.localScale = { x: 1, y: 1 };
        
        // 世界变换（缓存）
        this.worldMatrix = null;
        this.dirty = true;
    }
    
    // 添加子节点
    addChild(node) {
        if (node.parent) {
            node.parent.removeChild(node);
        }
        
        node.parent = this;
        this.children.push(node);
        node.markDirty();
        
        return this;
    }
    
    // 移除子节点
    removeChild(node) {
        const index = this.children.indexOf(node);
        if (index > -1) {
            this.children.splice(index, 1);
            node.parent = null;
        }
    }
    
    // 标记为脏
    markDirty() {
        this.dirty = true;
        this.children.forEach(child => child.markDirty());
    }
    
    // 获取世界变换
    getWorldMatrix() {
        if (!this.dirty && this.worldMatrix) {
            return this.worldMatrix;
        }
        
        const matrix = mat3.create();
        
        // 应用本地变换
        mat3.translate(matrix, matrix, [this.localPosition.x, this.localPosition.y]);
        mat3.rotate(matrix, matrix, this.localRotation);
        mat3.scale(matrix, matrix, [this.localScale.x, this.localScale.y]);
        
        // 应用父节点变换
        if (this.parent) {
            const parentMatrix = this.parent.getWorldMatrix();
            mat3.multiply(matrix, parentMatrix, matrix);
        }
        
        this.worldMatrix = matrix;
        this.dirty = false;
        
        return matrix;
    }
    
    // 遍历子节点
    traverse(callback) {
        callback(this);
        this.children.forEach(child => child.traverse(callback));
    }
    
    // 查找子节点
    find(name) {
        if (this.name === name) return this;
        
        for (const child of this.children) {
            const found = child.find(name);
            if (found) return found;
        }
        
        return null;
    }
}
```

```javascript
// 2D 变换矩阵工具
const mat3 = {
    create() {
        return new Float32Array([
            1, 0, 0,
            0, 1, 0,
            0, 0, 1
        ]);
    },
    
    translate(out, a, v) {
        const [x, y] = v;
        out[0] = a[0]; out[1] = a[1]; out[2] = a[2];
        out[3] = a[3]; out[4] = a[4]; out[5] = a[5];
        out[6] = a[0] * x + a[3] * y + a[6];
        out[7] = a[1] * x + a[4] * y + a[7];
        out[8] = a[2] * x + a[5] * y + a[8];
        return out;
    },
    
    rotate(out, a, rad) {
        const s = Math.sin(rad);
        const c = Math.cos(rad);
        const a00 = a[0], a01 = a[1], a02 = a[2];
        const a10 = a[3], a11 = a[4], a12 = a[5];
        
        out[0] = a00 * c + a10 * s;
        out[1] = a01 * c + a11 * s;
        out[2] = a02 * c + a12 * s;
        out[3] = a10 * c - a00 * s;
        out[4] = a11 * c - a01 * s;
        out[5] = a12 * c - a02 * s;
        out[6] = a[6]; out[7] = a[7]; out[8] = a[8];
        return out;
    },
    
    scale(out, a, v) {
        const [x, y] = v;
        out[0] = a[0] * x; out[1] = a[1] * x; out[2] = a[2] * x;
        out[3] = a[3] * y; out[4] = a[4] * y; out[5] = a[5] * y;
        out[6] = a[6]; out[7] = a[7]; out[8] = a[8];
        return out;
    },
    
    multiply(out, a, b) {
        const a00 = a[0], a01 = a[1], a02 = a[2];
        const a10 = a[3], a11 = a[4], a12 = a[5];
        const a20 = a[6], a21 = a[7], a22 = a[8];
        
        const b00 = b[0], b01 = b[1], b02 = b[2];
        const b10 = b[3], b11 = b[4], b12 = b[5];
        const b20 = b[6], b21 = b[7], b22 = b[8];
        
        out[0] = b00 * a00 + b01 * a10 + b02 * a20;
        out[1] = b00 * a01 + b01 * a11 + b02 * a21;
        out[2] = b00 * a02 + b01 * a12 + b02 * a22;
        out[3] = b10 * a00 + b11 * a10 + b12 * a20;
        out[4] = b10 * a01 + b11 * a11 + b12 * a21;
        out[5] = b10 * a02 + b11 * a12 + b12 * a22;
        out[6] = b20 * a00 + b21 * a10 + b22 * a20;
        out[7] = b20 * a01 + b21 * a11 + b22 * a21;
        out[8] = b20 * a02 + b21 * a12 + b22 * a22;
        return out;
    },
    
    transformPoint(out, a, p) {
        const [x, y] = p;
        out[0] = a[0] * x + a[3] * y + a[6];
        out[1] = a[1] * x + a[4] * y + a[7];
        return out;
    }
};
```

### 3.1 资源加载器
```javascript
class ResourceManager {
    constructor() {
        this.cache = new Map();
        this.loaders = new Map();
        this.progress = 0;
        this.total = 0;
        this.loaded = 0;
    }
    
    // 注册加载器
    registerLoader(type, loader) {
        this.loaders.set(type, loader);
    }
    
    // 加载单个资源
    async load(type, key, url) {
        // 检查缓存
        if (this.cache.has(key)) {
            return this.cache.get(key);
        }
        
        const loader = this.loaders.get(type);
        if (!loader) {
            throw new Error(`No loader registered for type: ${type}`);
        }
        
        this.total++;
        
        try {
            const resource = await loader.load(url);
            this.cache.set(key, resource);
            this.loaded++;
            this.progress = this.loaded / this.total;
            return resource;
        } catch (error) {
            console.error(`Failed to load ${key}:`, error);
            throw error;
        }
    }
    
    // 批量加载
    async loadAll(manifest) {
        const promises = Object.entries(manifest).map(([key, { type, url }]) =>
            this.load(type, key, url)
        );
        
        return Promise.all(promises);
    }
    
    // 获取资源
    get(key) {
        return this.cache.get(key);
    }
    
    // 卸载资源
    unload(key) {
        const resource = this.cache.get(key);
        if (resource && resource.destroy) {
            resource.destroy();
        }
        this.cache.delete(key);
    }
    
    // 清空缓存
    clear() {
        this.cache.forEach(resource => {
            if (resource.destroy) resource.destroy();
        });
        this.cache.clear();
    }
}

// 图片加载器
class ImageLoader {
    async load(url) {
        return new Promise((resolve, reject) => {
            const image = new Image();
            image.onload = () => resolve(image);
            image.onerror = reject;
            image.src = url;
        });
    }
}

class AudioLoader {
    constructor(audioContext) {
        this.context = audioContext;
    }
    
    async load(url) {
        const response = await fetch(url);
        const arrayBuffer = await response.arrayBuffer();
        return this.context.decodeAudioData(arrayBuffer);
    }
}

    }
}
```

```javascript
class TextureAtlas {
    constructor(image, data) {
        this.image = image;
        this.frames = new Map();
        
        // 解析图集数据
        this.parse(data);
    }
    
    parse(data) {
        // 支持多种格式
        if (data.frames) {
            // TexturePacker 格式
            Object.entries(data.frames).forEach(([name, frame]) => {
                this.frames.set(name, {
                    x: frame.frame.x,
                    y: frame.frame.y,
                    width: frame.frame.w,
                    height: frame.frame.h,
                    rotated: frame.rotated || false
                });
            });
        }
    }
    
    getFrame(name) {
        return this.frames.get(name);
    }
    
    draw(ctx, name, x, y, width, height) {
        const frame = this.getFrame(name);
        if (!frame) return;
        
        ctx.drawImage(
            this.image,
            frame.x, frame.y, frame.width, frame.height,
            x, y, width || frame.width, height || frame.height
        );
    }
}

// 精灵表动画
class SpriteSheetAnimation {
    constructor(atlas, frames, frameDuration = 0.1) {
        this.atlas = atlas;
        this.frames = frames;
        this.frameDuration = frameDuration;
        this.currentFrame = 0;
        this.elapsed = 0;
        this.playing = true;
        this.loop = true;
    }
    
    update(dt) {
        if (!this.playing) return;
        
        this.elapsed += dt;
        if (this.elapsed >= this.frameDuration) {
            this.elapsed = 0;
            this.currentFrame++;
            
            if (this.currentFrame >= this.frames.length) {
                if (this.loop) {
                    this.currentFrame = 0;
                } else {
                    this.currentFrame = this.frames.length - 1;
                    this.playing = false;
                }
            }
        }
    }
    
    draw(ctx, x, y, width, height) {
        const frameName = this.frames[this.currentFrame];
        this.atlas.draw(ctx, frameName, x, y, width, height);
    }
    
    play() {
        this.playing = true;
        this.currentFrame = 0;
        this.elapsed = 0;
    }
    
    stop() {
        this.playing = false;
    }
}
```

```javascript
class InputManager {
    constructor() {
        this.keys = new Map();
        this.mouse = {
            x: 0,
            y: 0,
            buttons: new Map()
        };
        this.touches = new Map();
        this.gamepads = new Map();
        
        this.setupKeyboard();
        this.setupMouse();
        this.setupTouch();
        this.setupGamepad();
    }
    
    // 键盘
    setupKeyboard() {
        window.addEventListener('keydown', (e) => {
            this.keys.set(e.code, {
                pressed: true,
                justPressed: true,
                timestamp: Date.now()
            });
        });
        
        window.addEventListener('keyup', (e) => {
            this.keys.set(e.code, {
                pressed: false,
                justReleased: true,
                timestamp: Date.now()
            });
        });
    }
    
    // 鼠标
    setupMouse() {
        window.addEventListener('mousemove', (e) => {
            this.mouse.x = e.clientX;
            this.mouse.y = e.clientY;
        });
        
        window.addEventListener('mousedown', (e) => {
            this.mouse.buttons.set(e.button, {
                pressed: true,
                justPressed: true,
                timestamp: Date.now()
            });
        });
        
        window.addEventListener('mouseup', (e) => {
            this.mouse.buttons.set(e.button, {
                pressed: false,
                justReleased: true,
                timestamp: Date.now()
            });
        });
    }
    
    // 触摸
    setupTouch() {
        window.addEventListener('touchstart', (e) => {
            Array.from(e.changedTouches).forEach(touch => {
                this.touches.set(touch.identifier, {
                    x: touch.clientX,
                    y: touch.clientY,
                    justPressed: true,
                    timestamp: Date.now()
                });
            });
        });
        
        window.addEventListener('touchmove', (e) => {
            Array.from(e.changedTouches).forEach(touch => {
                const existing = this.touches.get(touch.identifier);
                if (existing) {
                    existing.x = touch.clientX;
                    existing.y = touch.clientY;
                }
            });
        });
        
        window.addEventListener('touchend', (e) => {
            Array.from(e.changedTouches).forEach(touch => {
                this.touches.delete(touch.identifier);
            });
        });
    }
    
    // 手柄
    setupGamepad() {
        window.addEventListener('gamepadconnected', (e) => {
            this.gamepads.set(e.gamepad.index, {
                id: e.gamepad.id,
                buttons: [],
                axes: []
            });
        });
        
        window.addEventListener('gamepaddisconnected', (e) => {
            this.gamepads.delete(e.gamepad.index);
        });
    }
    
    // 更新（每帧调用）
    update() {
        // 清除 justPressed 和 justReleased
        this.keys.forEach((state, code) => {
            state.justPressed = false;
            state.justReleased = false;
        });
        
        this.mouse.buttons.forEach((state, button) => {
            state.justPressed = false;
            state.justReleased = false;
        });
        
        this.touches.forEach((state, id) => {
            state.justPressed = false;
        });
        
        // 更新手柄
        const gamepads = navigator.getGamepads();
        gamepads.forEach(gamepad => {
            if (gamepad) {
                this.gamepads.set(gamepad.index, {
                    id: gamepad.id,
                    buttons: gamepad.buttons.map(b => ({
                        pressed: b.pressed,
                        value: b.value
                    })),
                    axes: [...gamepad.axes]
                });
            }
        });
    }
    
    // 查询方法
    isKeyPressed(code) {
        return this.keys.get(code)?.pressed || false;
    }
    
    isKeyJustPressed(code) {
        return this.keys.get(code)?.justPressed || false;
    }
    
    isMouseButtonPressed(button = 0) {
        return this.mouse.buttons.get(button)?.pressed || false;
    }
    
    isMouseButtonJustPressed(button = 0) {
        return this.mouse.buttons.get(button)?.justPressed || false;
    }
    
    getMousePosition() {
        return { x: this.mouse.x, y: this.mouse.y };
    }
    
    getGamepadButton(index, button) {
        const gamepad = this.gamepads.get(index);
        return gamepad?.buttons[button]?.pressed || false;
    }
    
    getGamepadAxis(index, axis) {
        const gamepad = this.gamepads.get(index);
        return gamepad?.axes[axis] || 0;
    }
}
```

### 5.1 空间分区
```javascript
// 四叉树
class QuadTree {
    constructor(bounds, maxObjects = 10, maxLevels = 4, level = 0) {
        this.bounds = bounds;
        this.maxObjects = maxObjects;
        this.maxLevels = maxLevels;
        this.level = level;
        this.objects = [];
        this.nodes = [];
    }
    
    // 分裂
    split() {
        const { x, y, width, height } = this.bounds;
        const halfWidth = width / 2;
        const halfHeight = height / 2;
        
        this.nodes[0] = new QuadTree(
            { x: x + halfWidth, y, width: halfWidth, height: halfHeight },
            this.maxObjects, this.maxLevels, this.level + 1
        );
        this.nodes[1] = new QuadTree(
            { x, y, width: halfWidth, height: halfHeight },
            this.maxObjects, this.maxLevels, this.level + 1
        );
        this.nodes[2] = new QuadTree(
            { x, y: y + halfHeight, width: halfWidth, height: halfHeight },
            this.maxObjects, this.maxLevels, this.level + 1
        );
        this.nodes[3] = new QuadTree(
            { x: x + halfWidth, y: y + halfHeight, width: halfWidth, height: halfHeight },
            this.maxObjects, this.maxLevels, this.level + 1
        );
    }
    
    // 获取索引
    getIndex(rect) {
        const midX = this.bounds.x + this.bounds.width / 2;
        const midY = this.bounds.y + this.bounds.height / 2;
        
        const top = rect.y < midY && rect.y + rect.height < midY;
        const bottom = rect.y > midY;
        const left = rect.x < midX && rect.x + rect.width < midX;
        const right = rect.x > midX;
        
        if (top && right) return 0;
        if (top && left) return 1;
        if (bottom && left) return 2;
        if (bottom && right) return 3;
        
        return -1;
    }
    
    // 插入
    insert(rect) {
        if (this.nodes.length > 0) {
            const index = this.getIndex(rect);
            if (index !== -1) {
                this.nodes[index].insert(rect);
                return;
            }
        }
        
        this.objects.push(rect);
        
        if (this.objects.length > this.maxObjects && this.level < this.maxLevels) {
            if (this.nodes.length === 0) {
                this.split();
            }
            
            let i = 0;
            while (i < this.objects.length) {
                const index = this.getIndex(this.objects[i]);
                if (index !== -1) {
                    this.nodes[index].insert(this.objects.splice(i, 1)[0]);
                } else {
                    i++;
                }
            }
        }
    }
    
    // 查询
    retrieve(rect) {
        const result = [];
        
        if (this.nodes.length > 0) {
            const index = this.getIndex(rect);
            if (index !== -1) {
                result.push(...this.nodes[index].retrieve(rect));
            } else {
                this.nodes.forEach(node => {
                    result.push(...node.retrieve(rect));
                });
            }
        }
        
        result.push(...this.objects);
        return result;
    }
    
    // 清空
    clear() {
        this.objects = [];
        this.nodes.forEach(node => node.clear());
        this.nodes = [];
    }
}
```
