### 4.1 游戏 NPC AI
```python
# 使用 Q-Learning
import numpy as np

class QLearningAgent:
    def __init__(self, states, actions, lr=0.1, gamma=0.9, epsilon=0.1):
        self.q_table = np.zeros((states, actions))
        self.lr = lr
        self.gamma = gamma
        self.epsilon = epsilon
        self.actions = actions
    
    def choose_action(self, state):
        if np.random.uniform(0, 1) < self.epsilon:
            return np.random.choice(self.actions)
        else:
            return np.argmax(self.q_table[state, :])
    
    def learn(self, state, action, reward, next_state):
        predict = self.q_table[state, action]
        target = reward + self.gamma * np.max(self.q_table[next_state, :])
        self.q_table[state, action] += self.lr * (target - predict)

# 使用示例
agent = QLearningAgent(states=100, actions=4)

for episode in range(1000):
    state = env.reset()
    done = False
    
    while not done:
        action = agent.choose_action(state)
        next_state, reward, done, _ = env.step(action)
        agent.learn(state, action, reward, next_state)
        state = next_state
```

import numpy as np
from noise import pnoise2

# Perlin 噪声生成地形
def generate_terrain(width, height, scale=100):
    world = np.zeros((width, height))
    
    for i in range(width):
        for j in range(height):
            world[i][j] = pnoise2(i / scale, j / scale, octaves=6)
    
    return world

# 生成地图
terrain = generate_terrain(256, 256)

# 可视化
import matplotlib.pyplot as plt
plt.imshow(terrain, cmap='terrain')
plt.colorbar()
plt.show()

### 4.3 智能对话系统
```python
# 简单的对话系统
class DialogSystem:
    def __init__(self):
        self.context = {}
        self.rules = {
            "问候": ["你好！", "嗨！", "很高兴见到你！"],
            "告别": ["再见！", "下次见！", "保重！"],
            "询问": ["请问有什么可以帮助你的？", "你想了解什么？"]
        }
    
    def respond(self, user_input):
        # 简单的意图识别
        intent = self.classify_intent(user_input)
        
        # 获取回复
        responses = self.rules.get(intent, ["我不太明白你的意思。"])
        
        # 随机选择回复
        import random
        return random.choice(responses)
    
    def classify_intent(self, text):
        if any(word in text for word in ["你好", "嗨", "早上好"]):
            return "问候"
        elif any(word in text for word in ["再见", "拜拜", "下次见"]):
            return "告别"
        else:
            return "询问"

# 使用
dialog = DialogSystem()
print(dialog.respond("你好！"))
print(dialog.respond("什么是机器学习？"))
```
