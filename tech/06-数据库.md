### 2.2 DDL（数据定义）
```sql
-- 创建数据库
CREATE DATABASE IF NOT EXISTS mydb 
CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

USE mydb;

-- 创建表
CREATE TABLE users (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(50) NOT NULL UNIQUE,
    email VARCHAR(100) NOT NULL,
    password VARCHAR(255) NOT NULL,
    age INT DEFAULT 0,
    status TINYINT DEFAULT 1 COMMENT '0-禁用 1-启用',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_email (email),
    INDEX idx_status (status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户表';

-- 修改表结构
ALTER TABLE users ADD COLUMN phone VARCHAR(20);
ALTER TABLE users MODIFY COLUMN phone VARCHAR(30);
ALTER TABLE users DROP COLUMN phone;
ALTER TABLE users ADD INDEX idx_username (username);

-- 删除表
DROP TABLE IF EXISTS users;

-- 查看表结构
DESC users;
SHOW CREATE TABLE users;
```

### 2.5 多表查询
```sql
-- 内连接（交集）
SELECT u.username, o.order_no
FROM users u
INNER JOIN orders o ON u.id = o.user_id;

-- 左连接（左表全部）
SELECT u.username, o.order_no
FROM users u
LEFT JOIN orders o ON u.id = o.user_id;

-- 右连接（右表全部）
SELECT u.username, o.order_no
FROM users u
RIGHT JOIN orders o ON u.id = o.user_id;

-- 全连接（MySQL不支持，用UNION）
SELECT u.username, o.order_no FROM users u LEFT JOIN orders o ON u.id = o.user_id
UNION
SELECT u.username, o.order_no FROM users u RIGHT JOIN orders o ON u.id = o.user_id;

-- 自连接
SELECT e.name as employee, m.name as manager
FROM employees e
LEFT JOIN employees m ON e.manager_id = m.id;

-- 多表联查
SELECT u.username, p.title, c.content
FROM users u
JOIN posts p ON u.id = p.user_id
JOIN comments c ON p.id = c.post_id
WHERE u.status = 1;
```

### 2.6 窗口函数（MySQL 8.0+）
```sql
-- ROW_NUMBER() 行号
SELECT 
    username,
    age,
    ROW_NUMBER() OVER (ORDER BY age DESC) as rank_num
FROM users;

-- RANK() 排名（有并列会跳号）
SELECT 
    username,
    age,
    RANK() OVER (ORDER BY age DESC) as rank_num
FROM users;

-- DENSE_RANK() 排名（不跳号）
SELECT 
    username,
    age,
    DENSE_RANK() OVER (ORDER BY age DESC) as rank_num
FROM users;

-- 分组排名
SELECT 
    username,
    department,
    salary,
    RANK() OVER (PARTITION BY department ORDER BY salary DESC) as dept_rank
FROM employees;

-- 累计求和
SELECT 
    order_date,
    amount,
    SUM(amount) OVER (ORDER BY order_date) as cumulative_sum
FROM orders;

-- 移动平均
SELECT 
    order_date,
    amount,
    AVG(amount) OVER (ORDER BY order_date ROWS BETWEEN 2 PRECEDING AND CURRENT ROW) as moving_avg
FROM orders;
```

## 三、MySQL 进阶

### 3.1 索引优化
```sql
-- 索引类型
CREATE INDEX idx_name ON users(username);           -- 普通索引
CREATE UNIQUE INDEX idx_email ON users(email);      -- 唯一索引
CREATE INDEX idx_name_age ON users(username, age);  -- 联合索引
CREATE FULLTEXT INDEX idx_bio ON users(bio);        -- 全文索引

-- 索引使用原则
-- 1. 最左前缀原则
-- 联合索引 (a, b, c) 可以使用: a, ab, abc
-- 不能使用: b, c, bc

-- 2. 避免索引失效
-- 不要在索引列上使用函数
SELECT * FROM users WHERE YEAR(created_at) = 2024;  -- 错误
SELECT * FROM users WHERE created_at >= '2024-01-01'; -- 正确

-- 不要使用 != 或 NOT IN
-- 不要使用 LIKE '%xxx'
-- 不要在 OR 条件中混合索引和非索引列

-- 3. EXPLAIN 分析
EXPLAIN SELECT * FROM users WHERE username = 'zhangsan';
-- type: ALL(全表扫描) < index < range < ref < eq_ref < const
-- key: 使用的索引
-- rows: 扫描行数
```

### 3.2 事务
```sql
-- 开启事务
START TRANSACTION;

-- 或 BEGIN
BEGIN;

-- 执行SQL
UPDATE accounts SET balance = balance - 100 WHERE id = 1;
UPDATE accounts SET balance = balance + 100 WHERE id = 2;

-- 提交
COMMIT;

-- 回滚
ROLLBACK;

-- 事务特性（ACID）
-- A: 原子性 - 全部成功或全部失败
-- C: 一致性 - 数据前后一致
-- I: 隔离性 - 事务互不干扰
-- D: 持久性 - 提交后永久保存

-- 隔离级别
-- READ UNCOMMITTED - 可读取未提交（脏读）
-- READ COMMITTED - 只能读已提交（不可重复读）
-- REPEATABLE READ - 可重复读（MySQL默认）
-- SERIALIZABLE - 串行化（最安全但最慢）

-- 查看隔离级别
SELECT @@transaction_isolation;

-- 设置隔离级别
SET TRANSACTION ISOLATION LEVEL READ COMMITTED;
```

### 3.3 锁机制
```sql
-- 表锁
LOCK TABLES users READ;
LOCK TABLES users WRITE;
UNLOCK TABLES;

-- 行锁（InnoDB）
SELECT * FROM users WHERE id = 1 FOR UPDATE;  -- 排他锁
SELECT * FROM users WHERE id = 1 LOCK IN SHARE MODE;  -- 共享锁

-- 乐观锁（应用层实现）
-- 版本号方式
UPDATE products 
SET stock = stock - 1, version = version + 1 
WHERE id = 1 AND version = 1;

-- 死锁处理
-- 1. 查看死锁
SHOW ENGINE INNODB STATUS;
-- 2. 设置超时
SET innodb_lock_wait_timeout = 10;
```

### 3.4 存储过程
```sql
-- 创建存储过程
DELIMITER //
CREATE PROCEDURE GetUserById(IN userId BIGINT)
BEGIN
    SELECT * FROM users WHERE id = userId;
END //
DELIMITER ;

-- 调用
CALL GetUserById(1);

-- 带输出参数
DELIMITER //
CREATE PROCEDURE GetUserCount(OUT total INT)
BEGIN
    SELECT COUNT(*) INTO total FROM users;
END //
DELIMITER ;

CALL GetUserCount(@total);
SELECT @total;

-- 游标
DELIMITER //
CREATE PROCEDURE ProcessUsers()
BEGIN
    DECLARE done INT DEFAULT FALSE;
    DECLARE userId BIGINT;
    DECLARE userName VARCHAR(50);
    
    DECLARE cur CURSOR FOR SELECT id, username FROM users;
    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;
    
    OPEN cur;
    read_loop: LOOP
        FETCH cur INTO userId, userName;
        IF done THEN
            LEAVE read_loop;
        END IF;
        -- 处理逻辑
        SELECT CONCAT('Processing: ', userName);
    END LOOP;
    CLOSE cur;
END //
DELIMITER ;
```

## 七、数据库设计

### 7.1 范式
```
第一范式（1NF）：字段不可再分
  错误：地址 = "北京市朝阳区"
  正确：省 = "北京市", 区 = "朝阳区"

第二范式（2NF）：非主键字段完全依赖主键
  错误：(学号, 课程号) → 学生姓名
  正确：学号 → 学生姓名

第三范式（3NF）：非主键字段不传递依赖
  错误：学号 → 院系 → 院系电话
  正确：学号 → 院系，院系 → 院系电话
```

### 7.2 设计原则
```sql
-- 命名规范
-- 表名：小写 + 下划线，如 user_profiles
-- 字段名：小写 + 下划线，如 created_at
-- 主键：id 或 表名_id
-- 外键：关联表名_id
-- 布尔：is_ 开头，如 is_active
-- 时间：_at 结尾，如 created_at
-- 数量：_count 结尾，如 order_count

-- 必备字段
CREATE TABLE example (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at DATETIME NULL  -- 软删除
);
```

## 八、性能优化

### 8.1 查询优化
```sql
-- 慢查询日志
SET GLOBAL slow_query_log = ON;
SET GLOBAL long_query_time = 2;
SET GLOBAL slow_query_log_file = '/var/log/mysql/slow.log';

-- EXPLAIN 分析
EXPLAIN SELECT * FROM users WHERE age > 25;
-- 重点看：type, key, rows, Extra

-- 避免 SELECT *
SELECT id, username, email FROM users;

-- 避免在 WHERE 中使用函数
-- 错误
SELECT * FROM users WHERE DATE(created_at) = '2024-01-01';
-- 正确
SELECT * FROM users WHERE created_at >= '2024-01-01' AND created_at < '2024-01-02';

-- 使用 LIMIT 限制结果集
SELECT * FROM users LIMIT 100;
```

### 8.2 索引优化
```sql
-- 选择性高的列建索引
-- email（唯一）> status（2-3个值）

-- 联合索引顺序
-- 把选择性高的放前面
CREATE INDEX idx_email_status ON users(email, status);

-- 覆盖索引
-- 查询字段都在索引中，不需要回表
CREATE INDEX idx_covering ON users(username, email, age);
SELECT username, email, age FROM users WHERE username = '张三';

-- 避免过度索引
-- 每个索引都会占用空间，影响写入性能
```

## 十、常见面试题

### 1. MySQL 存储引擎区别
```
InnoDB：
- 支持事务、行锁、外键
- 支持 MVCC
- 聚簇索引
- 适合高并发读写

MyISAM：
- 不支持事务、行锁
- 表锁
- 非聚簇索引
- 适合读多写少
```

### 2. 事务隔离级别
```
READ UNCOMMITTED：脏读、不可重复读、幻读
READ COMMITTED：不可重复读、幻读
REPEATABLE READ：幻读（MySQL 默认，通过 MVCC 解决）
SERIALIZABLE：最安全但最慢
```

### 3. 索引数据结构
```
B+ 树（MySQL InnoDB）：
- 非叶子节点只存索引
- 叶子节点存数据，双向链表
- 适合范围查询和排序

Hash 索引：
- 等值查询 O(1)
- 不支持范围查询
- Memory 引擎默认

全文索引：
- 倒排索引
- 适合文本搜索
```

### 4. Redis 为什么快
```
1. 纯内存操作
2. 单线程避免上下文切换
3. I/O 多路复用
4. 高效数据结构（SDS、跳表、压缩列表）
```

### 5. 缓存与数据库一致性
```
Cache Aside Pattern：
1. 读：先缓存 → 未命中查DB → 写入缓存
2. 写：先更新DB → 再删除缓存

为什么删除而不是更新缓存？
- 避免并发写入导致脏数据
- 懒加载思想，需要时再重建
```
