### 1.1 脏矩形渲染
```javascript
class DirtyRectRenderer {
    constructor(canvas) {
        this.canvas = canvas;
        this.ctx = canvas.getContext('2d');
        this.dirtyRects = [];
        this.objects = [];
    }

    // 标记区域为脏
    markDirty(x, y, width, height) {
        this.dirtyRects.push({ x, y, width, height });
    }

    // 优化渲染
    render() {
        if (this.dirtyRects.length === 0) return;

        // 合并脏矩形
        const merged = this.mergeDirtyRects();

        // 只重绘脏区域
        merged.forEach(rect => {
            this.ctx.save();
            this.ctx.beginPath();
            this.ctx.rect(rect.x, rect.y, rect.width, rect.height);
            this.ctx.clip();

            // 清除区域
            this.ctx.clearRect(rect.x, rect.y, rect.width, rect.height);

            // 重绘该区域内的对象
            this.objects.forEach(obj => {
                if (this.intersects(obj, rect)) {
                    obj.render(this.ctx);
                }
            });

            this.ctx.restore();
        });

        this.dirtyRects = [];
    }

    // 合并重叠的脏矩形
    mergeDirtyRects() {
        // 简单合并算法
        const merged = [];
        const used = new Set();

        for (let i = 0; i < this.dirtyRects.length; i++) {
            if (used.has(i)) continue;

            let current = { ...this.dirtyRects[i] };

            for (let j = i + 1; j < this.dirtyRects.length; j++) {
                if (used.has(j)) continue;

                if (this.canMerge(current, this.dirtyRects[j])) {
                    current = this.merge(current, this.dirtyRects[j]);
                    used.add(j);
                }
            }

            merged.push(current);
        }

        return merged;
    }

    canMerge(a, b) {
        // 检查是否可以合并（有重叠或相邻）
        return !(a.x > b.x + b.width ||
                 b.x > a.x + a.width ||
                 a.y > b.y + b.height ||
                 b.y > a.y + a.height);
    }

    merge(a, b) {
        const x = Math.min(a.x, b.x);
        const y = Math.min(a.y, b.y);
        return {
            x,
            y,
            width: Math.max(a.x + a.width, b.x + b.width) - x,
            height: Math.max(a.y + a.height, b.y + b.height) - y
        };
    }

    intersects(obj, rect) {
        return !(obj.x > rect.x + rect.width ||
                 rect.x > obj.x + obj.width ||
                 obj.y > rect.y + rect.height ||
                 rect.y > obj.y + obj.height);
    }
}
```

```javascript
class OffscreenRenderer {
    constructor(width, height) {
        // 主Canvas
        this.mainCanvas = document.createElement('canvas');
        this.mainCanvas.width = width;
        this.mainCanvas.height = height;
        this.mainCtx = this.mainCanvas.getContext('2d');

        // 离屏Canvas（用于预渲染）
        this.offscreenCanvas = document.createElement('canvas');
        this.offscreenCanvas.width = width;
        this.offscreenCanvas.height = height;
        this.offscreenCtx = this.offscreenCanvas.getContext('2d');

        // 缓存
        this.cache = new Map();
    }

    // 预渲染到离屏Canvas
    prerender(key, renderFn) {
        const cached = this.cache.get(key);
        if (cached) return cached;

        // 创建新的离屏Canvas
        const offscreen = document.createElement('canvas');
        offscreen.width = this.mainCanvas.width;
        offscreen.height = this.mainCanvas.height;
        const ctx = offscreen.getContext('2d');

        // 渲染
        renderFn(ctx);

        // 缓存
        this.cache.set(key, offscreen);

        return offscreen;
    }

    // 绘制缓存的内容
    drawCached(key, x, y) {
        const cached = this.cache.get(key);
        if (cached) {
            this.mainCtx.drawImage(cached, x, y);
        }
    }

    // 清除缓存
    clearCache(key) {
        if (key) {
            this.cache.delete(key);
        } else {
            this.cache.clear();
        }
    }
}
```

### 1.3 对象池
```javascript
class ObjectPool {
    constructor(createFn, resetFn, initialSize = 100) {
        this.createFn = createFn;
        this.resetFn = resetFn;
        this.pool = [];
        this.active = [];

        // 预创建对象
        for (let i = 0; i < initialSize; i++) {
            this.pool.push(createFn());
        }
    }

    // 获取对象
    get() {
        let obj = this.pool.pop();

        if (!obj) {
            obj = this.createFn();
        }

        this.active.push(obj);
        return obj;
    }

    // 释放对象
    release(obj) {
        const index = this.active.indexOf(obj);
        if (index > -1) {
            this.active.splice(index, 1);
            this.resetFn(obj);
            this.pool.push(obj);
        }
    }

    // 释放所有
    releaseAll() {
        while (this.active.length > 0) {
            const obj = this.active.pop();
            this.resetFn(obj);
            this.pool.push(obj);
        }
    }

    // 获取活跃对象数
    get activeCount() {
        return this.active.length;
    }

    // 获取池中对象数
    get poolCount() {
        return this.pool.length;
    }
}

// 使用示例
const bulletPool = new ObjectPool(

    1000
```

```javascript
class Vector2 {
    constructor(x = 0, y = 0) {
        this.x = x;
        this.y = y;
    }

    // 加法
    add(v) {
        return new Vector2(this.x + v.x, this.y + v.y);
    }

    // 减法
    sub(v) {
        return new Vector2(this.x - v.x, this.y - v.y);
    }

    // 标量乘法
    multiply(scalar) {
        return new Vector2(this.x * scalar, this.y * scalar);
    }

    // 长度
    length() {
        return Math.sqrt(this.x * this.x + this.y * this.y);
    }

    // 归一化
    normalize() {
        const len = this.length();
        if (len === 0) return new Vector2(0, 0);
        return new Vector2(this.x / len, this.y / len);
    }

    // 点积
    dot(v) {
        return this.x * v.x + this.y * v.y;
    }

    // 距离
    distanceTo(v) {
        return this.sub(v).length();
    }

    // 角度
    angle() {
        return Math.atan2(this.y, this.x);
    }

    // 旋转
    rotate(angle) {
        const cos = Math.cos(angle);
        const sin = Math.sin(angle);
        return new Vector2(
            this.x * cos - this.y * sin,
            this.x * sin + this.y * cos
        );
    }

    // 克隆
    clone() {
        return new Vector2(this.x, this.y);
    }

    // 静态方法
    static fromAngle(angle, length = 1) {
        return new Vector2(
            Math.cos(angle) * length,
            Math.sin(angle) * length
        );
    }
}
```

### 2.2 碰撞检测
```javascript
class CollisionDetector {
    // 矩形碰撞
    static rectRect(a, b) {
        return a.x < b.x + b.width &&
               a.x + a.width > b.x &&
               a.y < b.y + b.height &&
               a.y + a.height > b.y;
    }

    // 圆形碰撞
    static circleCircle(a, b) {
        const dx = a.x - b.x;
        const dy = a.y - b.y;
        const distance = Math.sqrt(dx * dx + dy * dy);
        return distance < a.radius + b.radius;
    }

    // 矩形与圆形碰撞
    static rectCircle(rect, circle) {
        const closestX = Math.max(rect.x, Math.min(circle.x, rect.x + rect.width));
        const closestY = Math.max(rect.y, Math.min(circle.y, rect.y + rect.height));

        const dx = circle.x - closestX;
        const dy = circle.y - closestY;

        return (dx * dx + dy * dy) < (circle.radius * circle.radius);
    }

    // 点与矩形
    static pointRect(point, rect) {
        return point.x >= rect.x &&
               point.x <= rect.x + rect.width &&
               point.y >= rect.y &&
               point.y <= rect.y + rect.height;
    }

    // AABB碰撞
    static aabb(a, b) {
        return a.minX < b.maxX && a.maxX > b.minX &&
               a.minY < b.maxY && a.maxY > b.minY;
    }

    // 分离轴定理（SAT）
    static sat(shapeA, shapeB) {
        const axes = [...shapeA.getAxes(), ...shapeB.getAxes()];

        for (const axis of axes) {
            const projectionA = shapeA.project(axis);
            const projectionB = shapeB.project(axis);

            if (!this.overlap(projectionA, projectionB)) {
                return false; // 找到分离轴，无碰撞
            }
        }

        return true; // 所有轴都重叠，有碰撞
    }

    static overlap(a, b) {
        return a.min <= b.max && a.max >= b.min;
    }
}
```

### 2.3 物理模拟
```javascript
class PhysicsEngine {
    constructor() {
        this.bodies = [];
        this.gravity = new Vector2(0, 9.8);
        this.iterations = 10;
    }

    addBody(body) {
        this.bodies.push(body);
    }

    update(dt) {
        // 应用力
        this.bodies.forEach(body => {
            if (!body.isStatic) {
                // 重力
                body.velocity = body.velocity.add(this.gravity.multiply(dt));

                // 阻力
                body.velocity = body.velocity.multiply(1 - body.drag * dt);

                // 更新位置
                body.position = body.position.add(body.velocity.multiply(dt));
            }
        });

        // 碰撞检测和响应
        this.solveCollisions();
    }

    solveCollisions() {
        for (let i = 0; i < this.bodies.length; i++) {
            for (let j = i + 1; j < this.bodies.length; j++) {
                const a = this.bodies[i];
                const b = this.bodies[j];

                if (CollisionDetector.rectRect(a, b)) {
                    this.resolveCollision(a, b);
                }
            }
        }
    }

    resolveCollision(a, b) {
        // 计算碰撞法线和深度
        const dx = (a.x + a.width / 2) - (b.x + b.width / 2);
        const dy = (a.y + a.height / 2) - (b.y + b.height / 2);

        const width = (a.width + b.width) / 2;
        const height = (a.height + b.height) / 2;

        const crossWidth = width * dy;
        const crossHeight = height * dx;

        let normal;
        let penetration;

        if (crossWidth > crossHeight) {
            if (crossWidth > -crossHeight) {
                normal = new Vector2(0, 1);
                penetration = (a.y + a.height) - b.y;
            } else {
                normal = new Vector2(-1, 0);
                penetration = (b.x + b.width) - a.x;
            }
        } else {
            if (crossWidth > -crossHeight) {
                normal = new Vector2(1, 0);
                penetration = (a.x + a.width) - b.x;
            } else {
                normal = new Vector2(0, -1);
                penetration = (b.y + b.height) - a.y;
            }
        }

        // 分离物体
        if (!a.isStatic) {
            a.position = a.position.add(normal.multiply(penetration / 2));
        }
        if (!b.isStatic) {
            b.position = b.position.sub(normal.multiply(penetration / 2));
        }

        // 计算相对速度
        const relativeVelocity = a.velocity.sub(b.velocity);
        const velocityAlongNormal = relativeVelocity.dot(normal);

        // 不处理分离
        if (velocityAlongNormal > 0) return;

        // 计算恢复系数
        const restitution = Math.min(a.restitution, b.restitution);

        // 计算冲量
        let j = -(1 + restitution) * velocityAlongNormal;
        j /= (1 / a.mass) + (1 / b.mass);

        // 应用冲量
        const impulse = normal.multiply(j);
        a.velocity = a.velocity.add(impulse.multiply(1 / a.mass));
        b.velocity = b.velocity.sub(impulse.multiply(1 / b.mass));
    }
}

class PhysicsBody {
    constructor(options = {}) {
        this.position = options.position || new Vector2(0, 0);
        this.velocity = options.velocity || new Vector2(0, 0);
        this.acceleration = options.acceleration || new Vector2(0, 0);
        this.mass = options.mass || 1;
        this.restitution = options.restitution || 0.5;
        this.drag = options.drag || 0.01;
        this.isStatic = options.isStatic || false;

        // 碰撞体
        this.width = options.width || 0;
        this.height = options.height || 0;
        this.radius = options.radius || 0;
    }

    applyForce(force) {
        this.velocity = this.velocity.add(force.multiply(1 / this.mass));
    }
}
```

```javascript
class Tween {
    constructor(target, properties, duration, easing = 'linear') {
        this.target = target;
        this.properties = properties;
        this.duration = duration;
        this.easing = EasingFunctions[easing] || EasingFunctions.linear;

        this.startValues = {};
        this.currentTime = 0;
        this.isPlaying = false;
        this.onComplete = null;
        this.onUpdate = null;

        // 保存初始值
        for (const key in properties) {
            this.startValues[key] = target[key];
        }
    }

    start() {
        this.isPlaying = true;
        this.currentTime = 0;
        return this;
    }

    update(dt) {
        if (!this.isPlaying) return;

        this.currentTime += dt;
        const progress = Math.min(this.currentTime / this.duration, 1);
        const easedProgress = this.easing(progress);

        // 更新属性
        for (const key in this.properties) {
            const start = this.startValues[key];
            const end = this.properties[key];
            this.target[key] = start + (end - start) * easedProgress;
        }

        // 回调
        if (this.onUpdate) {
            this.onUpdate(progress);
        }

        // 完成
        if (progress >= 1) {
            this.isPlaying = false;
            if (this.onComplete) {
                this.onComplete();
            }
        }

        return this;
    }

    stop() {
        this.isPlaying = false;
        return this;
    }

    reset() {
        this.currentTime = 0;
        for (const key in this.startValues) {
            this.target[key] = this.startValues[key];
        }
        return this;
    }
}

// 缓动函数
const EasingFunctions = {
    linear: t => t,
    easeInQuad: t => t * t,
    easeOutQuad: t => t * (2 - t),
    easeInOutQuad: t => t < 0.5 ? 2 * t * t : -1 + (4 - 2 * t) * t,
    easeInCubic: t => t * t * t,
    easeOutCubic: t => (--t) * t * t + 1,
    easeInOutCubic: t => t < 0.5 ? 4 * t * t * t : (t - 1) * (2 * t - 2) * (2 * t - 2) + 1,
    easeInElastic: t => {
        if (t === 0 || t === 1) return t;
        return -Math.pow(2, 10 * (t - 1)) * Math.sin((t - 1.1) * 5 * Math.PI);
    },
    easeOutElastic: t => {
        if (t === 0 || t === 1) return t;
        return Math.pow(2, -10 * t) * Math.sin((t - 0.1) * 5 * Math.PI) + 1;
    },
    easeInBounce: t => 1 - EasingFunctions.easeOutBounce(1 - t),
    easeOutBounce: t => {
        if (t < 1 / 2.75) return 7.5625 * t * t;
        if (t < 2 / 2.75) return 7.5625 * (t -= 1.5 / 2.75) * t + 0.75;
        if (t < 2.5 / 2.75) return 7.5625 * (t -= 2.25 / 2.75) * t + 0.9375;
        return 7.5625 * (t -= 2.625 / 2.75) * t + 0.984375;
    }
};

// 动画管理器
class TweenManager {
    constructor() {
        this.tweens = [];
    }

    create(target, properties, duration, easing) {
        const tween = new Tween(target, properties, duration, easing);
        this.tweens.push(tween);
        return tween;
    }

    update(dt) {
        this.tweens = this.tweens.filter(tween => {
            tween.update(dt);
            return tween.isPlaying;
        });
    }

    clear() {
        this.tweens = [];
    }
}
```

### 3.2 粒子系统
```javascript
class ParticleSystem {
    constructor(options = {}) {
        this.particles = [];
        this.emitters = [];
        this.maxParticles = options.maxParticles || 1000;
    }

    addEmitter(emitter) {
        this.emitters.push(emitter);
    }

    update(dt) {
        // 发射新粒子
        this.emitters.forEach(emitter => {
            const count = emitter.getEmitCount(dt);
            for (let i = 0; i < count; i++) {
                if (this.particles.length < this.maxParticles) {
                    this.particles.push(emitter.createParticle());
                }
            }
        });

        // 更新粒子
        this.particles = this.particles.filter(particle => {
            particle.update(dt);
            return particle.isAlive;
        });
    }

    render(ctx) {
        this.particles.forEach(particle => {
            particle.render(ctx);
        });
    }
}

class ParticleEmitter {
    constructor(options = {}) {
        this.x = options.x || 0;
        this.y = options.y || 0;
        this.rate = options.rate || 10; // 每秒发射数
        this.lifetime = options.lifetime || 1;
        this.speed = options.speed || 100;
        this.spread = options.spread || Math.PI * 2;
        this.direction = options.direction || 0;

        this.accumulator = 0;
    }

    getEmitCount(dt) {
        this.accumulator += this.rate * dt;
        const count = Math.floor(this.accumulator);
        this.accumulator -= count;
        return count;
    }

    createParticle() {
        const angle = this.direction + (Math.random() - 0.5) * this.spread;
        const speed = this.speed * (0.5 + Math.random() * 0.5);

        return new Particle({
            x: this.x,
            y: this.y,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed,
            lifetime: this.lifetime * (0.8 + Math.random() * 0.4),
            size: 2 + Math.random() * 4,
            color: this.getColor()
        });
    }

    getColor() {
        const colors = ['#ff0000', '#ff6600', '#ffcc00', '#ffffff'];
        return colors[Math.floor(Math.random() * colors.length)];
    }
}

class Particle {
    constructor(options = {}) {
        this.x = options.x || 0;
        this.y = options.y || 0;
        this.vx = options.vx || 0;
        this.vy = options.vy || 0;
        this.lifetime = options.lifetime || 1;
        this.maxLifetime = this.lifetime;
        this.size = options.size || 2;
        this.color = options.color || '#ffffff';
        this.alpha = 1;
        this.isAlive = true;
    }

    update(dt) {
        this.x += this.vx * dt;
        this.y += this.vy * dt;
        this.lifetime -= dt;

        // 淡出
        this.alpha = Math.max(0, this.lifetime / this.maxLifetime);

        // 缩小
        this.size *= 0.99;

        if (this.lifetime <= 0 || this.size < 0.1) {
            this.isAlive = false;
        }
    }

    render(ctx) {
        ctx.save();
        ctx.globalAlpha = this.alpha;
        ctx.fillStyle = this.color;
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
        ctx.fill();
        ctx.restore();
    }
}
```

```javascript
class StateMachine {
    constructor() {
        this.states = new Map();
        this.currentState = null;
        this.previousState = null;
    }

    addState(name, state) {
        this.states.set(name, state);
        state.name = name;
        state.fsm = this;
    }

    setState(name) {
        const newState = this.states.get(name);
        if (!newState) return;

        // 退出当前状态
        if (this.currentState) {
            this.currentState.exit?.();
        }

        // 保存上一个状态
        this.previousState = this.currentState;

        // 进入新状态
        this.currentState = newState;
        this.currentState.enter?.();
    }

    update(dt) {
        this.currentState?.update?.(dt);
    }

    revertToPrevious() {
        if (this.previousState) {
            this.setState(this.previousState.name);
        }
    }
}

class State {
    constructor(options = {}) {
        this.name = '';
        this.fsm = null;
        this.enter = options.enter;
        this.update = options.update;
        this.exit = options.exit;
    }
}
```

```javascript
class BehaviorTree {
    constructor(root) {
        this.root = root;
    }

    update(dt) {
        return this.root.execute(dt);
    }

class BTNode {
    execute(dt) {
        return 'success'; // 'success', 'failure', 'running'
    }
}

class BTSequence extends BTNode {
    constructor(children) {
        super();
        this.children = children;
        this.currentIndex = 0;
    }

    execute(dt) {
        for (let i = this.currentIndex; i < this.children.length; i++) {
            const result = this.children[i].execute(dt);

            if (result === 'failure') {
                this.currentIndex = 0;
                return 'failure';
            }

            if (result === 'running') {
                this.currentIndex = i;
                return 'running';
            }
        }

        this.currentIndex = 0;
        return 'success';
    }
}

                return 'success';
            }

            if (result === 'running') {
                this.currentIndex = i;

    }

    execute(dt) {
        return this.conditionFn() ? 'success' : 'failure';
    }
}

class BTAction extends BTNode {
    constructor(actionFn) {
        super();
        this.actionFn = actionFn;
    }

    execute(dt) {
        return this.actionFn(dt);
    }
}

            new BTCondition(() => enemyNearby()),

            new BTCondition(() => healthLow()),
            new BTAction((dt) => findHealth(dt))
```

### 5.1 音频管理器
```javascript
class AudioManager {
    constructor() {
        this.context = new (window.AudioContext || window.webkitAudioContext)();
        this.sounds = new Map();
        this.music = null;
        this.masterVolume = 1;
        this.sfxVolume = 1;
        this.musicVolume = 0.5;
    }

    async loadSound(name, url) {
        const response = await fetch(url);
        const arrayBuffer = await response.arrayBuffer();
        const audioBuffer = await this.context.decodeAudioData(arrayBuffer);
        this.sounds.set(name, audioBuffer);
    }

    playSound(name, options = {}) {
        const buffer = this.sounds.get(name);
        if (!buffer) return;

        const source = this.context.createBufferSource();
        source.buffer = buffer;

        // 音量控制
        const gainNode = this.context.createGain();
        gainNode.gain.value = (options.volume || 1) * this.sfxVolume * this.masterVolume;

        // 连接
        source.connect(gainNode);
        gainNode.connect(this.context.destination);

        // 播放
        source.start(0);

        return source;
    }

    async playMusic(url, loop = true) {
        // 停止当前音乐
        this.stopMusic();

        const response = await fetch(url);
        const arrayBuffer = await response.arrayBuffer();
        const audioBuffer = await this.context.decodeAudioData(arrayBuffer);

        this.music = this.context.createBufferSource();
        this.music.buffer = audioBuffer;
        this.music.loop = loop;

        // 音量控制
        const gainNode = this.context.createGain();
        gainNode.gain.value = this.musicVolume * this.masterVolume;

        this.music.connect(gainNode);
        gainNode.connect(this.context.destination);

        this.music.start(0);
    }

    stopMusic() {
        if (this.music) {
            this.music.stop();
            this.music = null;
        }
    }

    setMasterVolume(volume) {
        this.masterVolume = Math.max(0, Math.min(1, volume));
    }

    setSFXVolume(volume) {
        this.sfxVolume = Math.max(0, Math.min(1, volume));
    }

    setMusicVolume(volume) {
        this.musicVolume = Math.max(0, Math.min(1, volume));
    }
}
```
