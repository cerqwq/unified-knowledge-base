### 1.2 Vue 3 响应式原理
```javascript
// Proxy 实现
const handler = {
    get(target, key, receiver) {
        track(target, key)  // 收集依赖
        return Reflect.get(target, key, receiver)
    },
    set(target, key, value, receiver) {
        const result = Reflect.set(target, key, value, receiver)
        trigger(target, key)  // 触发更新
        return result
    }
}

function reactive(obj) {
    return new Proxy(obj, handler)
}

// effect 自动追踪
let activeEffect = null
function effect(fn) {
    activeEffect = fn
    fn()  // 执行时触发 get，收集依赖
    activeEffect = null
}

// 依赖收集
const targetMap = new WeakMap()
function track(target, key) {
    if (!activeEffect) return
    
    let depsMap = targetMap.get(target)
    if (!depsMap) {
        depsMap = new Map()
        targetMap.set(target, depsMap)
    }
    
    let deps = depsMap.get(key)
    if (!deps) {
        deps = new Set()
        depsMap.set(key, deps)
    }
    
    deps.add(activeEffect)
}

// 触发更新
function trigger(target, key) {
    const depsMap = targetMap.get(target)
    if (!depsMap) return
    
    const deps = depsMap.get(key)
    if (deps) {
        deps.forEach(effect => effect())
    }
}
```

### 2.1 React 设计理念
```
核心思想：
1. UI = f(state)  — 视图是状态的函数

3. 组合优于继承 — 用组合构建 UI
4. 声明式 — 描述"是什么"，而不是"怎么做"

架构：
- 协调（Reconciliation）— Virtual DOM diff
- 调度（Scheduler）— 时间分片
- 渲染器（Renderer）— DOM、Native、SSR
```

### 2.2 React Fiber 架构
```javascript
// Fiber 节点结构
{
    tag: FunctionComponent,  // 组件类型
    type: MyComponent,       // 组件函数/类
    key: null,
    stateNode: null,
    
    // Fiber 树
```

### 2.3 Hooks 原理
```javascript
// hooks 存储在 fiber.memoizedState 链表中
let workInProgressHook = null
let currentHook = null

function mountState(initialState) {
    const hook = {
        memoizedState: initialState,
        queue: { pending: null },
        next: null
    }
    
    // 链接到 fiber
    if (workInProgressHook === null) {
        currentlyRenderingFiber.memoizedState = hook
        workInProgressHook = hook
    } else {
        workInProgressHook.next = hook
        workInProgressHook = hook
    }
    
    return [hook.memoizedState, dispatchAction.bind(null, hook.queue)]
}

function updateState() {
    const hook = updateWorkInProgressHook()
    return [hook.memoizedState, dispatchAction.bind(null, hook.queue)]
}

// useState 实现
function useState(initialState) {
    if (isMount) {
        return mountState(initialState)
    } else {
        return updateState()
    }
}

// useEffect 实现
function useEffect(callback, deps) {
    const hook = updateWorkInProgressHook()
    
    if (currentHook) {
        const prevEffect = currentHook.memoizedState
        if (areDepsEqual(deps, prevEffect.deps)) {
            return  // 依赖没变，跳过
        }
    }
    
    hook.memoizedState = {
        callback,
        deps,
        destroy: null
    }
    
    // 注册到 effect 列表
    schedulePassiveEffect(hook)
}
```

### 2.5 Redux 设计思想

```javascript
// 三大原则
// 1. 单一数据源
const store = {
    todos: [],
    user: {},
    ui: {}
}

// 2. State 只读
// 只能通过 action 修改

// 3. 纯函数修改
function todosReducer(state = [], action) {
    switch (action.type) {
        case 'todos/add':
            return [...state, { text: action.payload, done: false }]
        case 'todos/toggle':
            return state.map((todo, i) =>
                i === action.index ? { ...todo, done: !todo.done } : todo
            )
        default:
            return state
    }
}

            state.push({ text: action.payload, done: false })

            state[action.index].done = !state[action.index].done
        }
    }
```

## 三、Rich Harris — Svelte 创造者

### 3.1 Svelte 核心理念
```
"编译时框架"思想：
- 传统框架：运行时框架代码 + 业务代码
- Svelte：编译时把框架代码消除，只输出业务代码

优势：
- 更小的包体积（无运行时）
- 更好的性能（直接操作 DOM）
- 更简单的语法（原生 JS）
```

### 3.2 Svelte 语法

## 五、前端性能优化标杆

### 5.1 Web Vitals 指标
```
Core Web Vitals:
1. LCP (Largest Contentful Paint) — 最大内容绘制
   - 目标：< 2.5s
   - 优化：预加载、优化图片、CDN

2. FID (First Input Delay) — 首次输入延迟
   - 目标：< 100ms
   - 优化：减少主线程阻塞、代码分割

3. CLS (Cumulative Layout Shift) — 累积布局偏移
   - 目标：< 0.1
   - 优化：设置图片尺寸、避免动态插入内容

其他指标：
- TTFB (Time to First Byte) — 首字节时间
- FCP (First Contentful Paint) — 首次内容绘制
- TTI (Time to Interactive) — 可交互时间
```
